    private void validateSettings() {
        int selectedMode = modeCombo.getSelectedIndex();
        
        logArea.append("\n=== AYAR DOĞRULAMA ===\n");
        
        // Seed kontrolü
        String seedText = seedField.getText().trim();
        if (selectedMode != 1) { // Gerçek dünya modu hariç
            if (seedText.isEmpty()) {
                logArea.append("⚠ Seed boş! Rastgele seed oluşturuluyor...\n");
                generateRandomSeed();
            } else {
                try {
                    long seed = Long.parseLong(seedText);
                    logArea.append("✓ Seed geçerli: " + seed + "\n");
                } catch (NumberFormatException e) {
                    long seed = seedText.hashCode();
                    logArea.append("✓ String seed hash'e çevrildi: " + seed + "\n");
                }
            }
        }
        
        // Mod kontrolü
        String[] modeNames = {"Seed Simülasyonu", "Gerçek Dünya", "Hibrit Mod"};
        logArea.append("✓ Mod: " + modeNames[selectedMode] + "\n");
        
        // Dünya dizini kontrolü - detaylı
        if (selectedMode == 1 || selectedMode == 2) {
            logArea.append("\n--- DÜNYA DİZİNİ KONTROLÜ ---\n");
            
            if (selectedWorldDirectory != null) {
                // Detaylı dünya analizi
                String worldInfo = RealSeedAnalyzer.getWorldDirectoryInfo(selectedWorldDirectory);
                logArea.append(worldInfo);
                
                // Geçerlilik kontrolü
                boolean isValid = RealSeedAnalyzer.isValidWorldDirectory(selectedWorldDirectory);
                if (isValid) {
                    logArea.append("✓ Dünya dosyası analiz için uygun\n");
                } else {
                    logArea.append("✗ Dünya dosyası geçersiz!\n");
                    logArea.append("  Gereksinimler:\n");
                    logArea.append("  - 'region' klasörü olmalı\n");
                    logArea.append("  - En az bir .mca dosyası bulunmalı\n");
                    logArea.append("  - Dosyalar okunabilir olmalı\n");
                }
            } else {
                if (selectedMode == 1) {
                    logArea.append("✗ Gerçek dünya modu için dünya dizini gerekli!\n");
                    logArea.append("  Yapılacaklar:\n");
                    logArea.append("  1. 'Gözat' butonuna tıklayın\n");
                    logArea.append("  2. .minecraft/saves klasöründen dünya seçin\n");
                    logArea.append("  3. Dünya klasörünün içinde 'region' olduğundan emin olun\n");
                } else {
                    logArea.append("⚠ Hibrit modda dünya dizini opsiyonel\n");
                }
            }
            logArea.append("-----------------------------\n");
        }
        
        // Arama alanı kontrolü
        int startX = (int) startXSpinner.getValue();
        int endX = (int) endXSpinner.getValue();
        int startZ = (int) startZSpinner.getValue(); 
        int endZ = (int) endZSpinner.getValue();
        int minY = (int) minYSpinner.getValue();
        int maxY = (int) maxYSpinner.getValue();
        
        int areaSize = (endX - startX + 1) * (endZ - startZ + 1);
        int layers = maxY - minY + 1;
        
        logArea.append("✓ Arama alanı: " + areaSize + " blok (" + (endX-startX+1) + "x" + (endZ-startZ+1) + ")\n");
        logArea.append("✓ Y katmanları: " + layers + " katman (" + minY + " → " + maxY + ")\n");
        logArea.append("✓ Toplam arama hacmi: " + (areaSize * layers) + " blok\n");
        
        // Performans tahmini
        if (areaSize > 1000000) {
            logArea.append("⚠ Büyük arama alanı - işlem uzun sürebilir\n");
        } else if (areaSize > 100000) {
            logArea.append("⚠ Orta büyüklükte arama alanı - birkaç dakika sürebilir\n");
        } else {
            logArea.append("✓ Küçük arama alanı - hızlı sonuç beklenir\n");
        }
        
        logArea.append("===================\n\n");
    }
